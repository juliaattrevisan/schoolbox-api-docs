/* Summators
 */
$allSummators = [
    new Summator\Average(),
    new Summator\Maximum(),
    new Summator\Minimum(),
    new Summator\Sum()
];

echo "<h1>Summators</h1>";

echo "<h1>Types and Visibility</h1>";

new \Alaress_Probe('Title', 'Type', 'Accept Online?', 'Accepts Multi?', 'Show Unsubmitted?', 'Has Rubric?');

foreach ($project->getAssessables() as $currentAssessable) {
    if ($currentAssessable instanceof DueWork) {
        new \Alaress_Probe(
            $currentAssessable->getTitle(),
            $currentAssessable->getTypeName(),
            $currentAssessable->getAcceptOnline(),
            $currentAssessable->isAcceptMulti(),
            !$currentAssessable->getShowSubmittedOnly(),
            $currentAssessable->hasRubric()
        );
    }
    else {
        new \Alaress_Probe($currentAssessable->getTitle(), $currentAssessable->getTypeName(), '', '', '', '');
    }
}

new \Alaress_Probe($project->getTypeName());


echo "<h1>Submission</h1>";

new \Alaress_Probe('Title', 'Files', 'Plagscan', 'Submitted By', 'Submitted On Behalf Of', 'Submitted At');

foreach ($project->getAssessables() as $currentAssessable) {
    if ($currentAssessable instanceof DueWork) {

        /** @var DueWorkSubmissionFile $currentFile */
        /** @var \Schoolbox\User $submittedBy */
        /** @var \Schoolbox\User $submittedOnBehalfOf */

        $latestSubmission = null;
        $submittedBy = null;
        $submittedOnBehalfOf = null;
        $allFiles = [];
        $allPlagscan = [];

        if ($currentAssessable->getAllResponses()->count()) {
            $latestSubmission = $currentAssessable->getAllResponses()->first();
            $submittedBy = $latestSubmission->getSubmittedBy();
            $submittedOnBehalfOf = $latestSubmission->getSubmittedOnBehalfOf();


            if ($latestSubmission && $latestSubmission->getAllFiles()->count()) {
                foreach ($latestSubmission->getAllFiles() as $currentFile) {
                    $allFiles[] = $currentFile->getFilename();
                    $allPlagscan[] = $currentFile->findPlagscanScore();
                }
            }
        }

        new \Alaress_Probe(
            $currentAssessable->getTitle(),
            implode('<br />', $allFiles),
            implode('<br />', $allPlagscan),
            $submittedBy ? $submittedBy->getFullname() : '',
            $submittedOnBehalfOf ? $submittedOnBehalfOf->getFullname() : '',
            $latestSubmission ? $latestSubmission->getSubmittedAt()->getLocal() : ''
        );
    }
}


echo "<h1>Feedbacks</h1>";

new \Alaress_Probe('Title', 'Files', 'Mark', 'Comment', 'Published At', 'Published?');

foreach ($project->getAssessables() as $currentAssessable) {
    if ($currentAssessable instanceof DueWork) {

        /** @var DueWorkSubmissionFile $currentFile */
        /** @var \Schoolbox\User $submittedBy */
        /** @var \Schoolbox\User $submittedOnBehalfOf */
        /** @var DueWorkFeedback $latestFeedback */

        $latestSubmission = null;
        $submittedBy = null;
        $submittedOnBehalfOf = null;
        $allFiles = [];
        $allPlagscan = [];

        $latestFeedback = null;

        if ($currentAssessable->getAllResponses()->count()) {
            $latestSubmission = $currentAssessable->getAllResponses()->first();
            $submittedBy = $latestSubmission->getSubmittedBy();
            $submittedOnBehalfOf = $latestSubmission->getSubmittedOnBehalfOf();

            $latestFeedback = $latestSubmission->findMostRecentNonNullFeedback();

            if ($latestFeedback) {
                foreach ($latestFeedback->getAllFiles() as $currentFile) {
                    $allFiles[] = $currentFile->getFilename();
                }
            }
        }

        new \Alaress_Probe(
            $currentAssessable->getTitle(),
            implode('<br />', $allFiles),
            $latestFeedback ? $latestFeedback->getMark() : '',
            $latestFeedback ? $latestFeedback->getComment() : '',
            $latestFeedback ? $latestFeedback->getPublishedAt()->getLocal() : '',
            $latestFeedback ? 'Published' : 'Unpublished'
        );
    }
}
