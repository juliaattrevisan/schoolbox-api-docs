// Assessment Summary by hand
//
// -3 + 25 + 19 [5,4,5,5] + Completed
//
// => 41/3 = 13.67 (Average)

// => 41 (Sum)
//
// 4/5 things marked, or completed


/* Summators
 */
$allSummators = [
    new Summator\Average(),
    new Summator\Maximum(),
    new Summator\Minimum(),
    new Summator\Sum()
];

echo "<h1>Summators</h1>";
foreach ($allSummators as $currentSummator) {
    $project = new Project($currentSummator);
    $project->getAssessables()
        ->add($dueWork1)
        ->add($dueWork2)
        ->add($dueWork3)
        ->add($quiz1)
        ->add($task1);

    new \Alaress_Probe(get_class($currentSummator), $project->calculateMark(), $project->calculatePercentageMarked());
}

echo "<h1>Types and Visibility</h1>";


new \Alaress_Probe('Type', 'Accept Online?', 'Show Unsubmitted?', 'Has Rubric?');

foreach ($project->getAssessables() as $currentAssessable) {

    if ($currentAssessable instanceof DueWork) {
        new \Alaress_Probe($currentAssessable->getTypeName(), $currentAssessable->getAcceptOnline(), !$currentAssessable->getShowSubmittedOnly(), $currentAssessable->hasRubric());
    }
    else {
        new \Alaress_Probe($currentAssessable->getTypeName());
    }
}

new \Alaress_Probe($project->getTypeName());
