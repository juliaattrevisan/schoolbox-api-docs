        new \Alaress_Probe(

            $currentAssessable->getAcceptOnline(),
            $currentAssessable->isAcceptMulti(),
            !$currentAssessable->getShowSubmittedOnly(),

echo "<h1>Submission</h1>";

new \Alaress_Probe('Title', 'Files', 'Plagscan', 'Submitted By', 'Submitted On Behalf Of', 'Submitted At');

foreach ($project->getAssessables() as $currentAssessable) {

    /** @var \Schoolbox\User $submittedBy */
    $submittedBy = null;

    if ($currentAssessable instanceof DueWork) {

        /** @var DueWorkSubmission $latestSubmission */
        $latestSubmission = null;

        /** @var \Schoolbox\User $submittedOnBehalfOf */
        $submittedOnBehalfOf = null;

        /* Plagscan docs
         */
        $allPlagscan = [];

        /* Attached files
         */
        $allFiles = [];

        if ($currentAssessable->getAllResponses()->count()) {
            $latestSubmission = $currentAssessable->getAllResponses()->first();
            $submittedBy = $latestSubmission->getSubmittedBy();
            $submittedOnBehalfOf = $latestSubmission->getSubmittedOnBehalfOf();

            /** @var DueWorkSubmissionFile $currentFile */
            if ($latestSubmission && $latestSubmission->getAllFiles()->count()) {
                foreach ($latestSubmission->getAllFiles() as $currentFile) {
                    $allFiles[] = $currentFile->getFilename();
                    $allPlagscan[] = $currentFile->findPlagscanScore();
                }
            }
        }

        new \Alaress_Probe(
            $currentAssessable->getTitle(),
            implode('<br />', $allFiles),
            implode('<br />', $allPlagscan),
            $submittedBy ? $submittedBy->getFullname() : '',
            $submittedOnBehalfOf ? $submittedOnBehalfOf->getFullname() : '',
            $latestSubmission && $latestSubmission ? $latestSubmission->getSubmittedAt()->getLocal() : ''
        );
    }
    elseif ($currentAssessable instanceof Quiz) {

        /** @var QuizAttempt $latestSubmission */
        $latestSubmission = null;

        if ($currentAssessable->getAllResponses()->count()) {
            $latestSubmission = $currentAssessable->getAllResponses()->first();
            $submittedBy = $latestSubmission->getSubmittedBy();
        }

        new \Alaress_Probe(
            $currentAssessable->getTitle(),
            '',
            '',
            $submittedBy ? $submittedBy->getFullname() : '',
            '',
            $latestSubmission && $latestSubmission ? $latestSubmission->getSubmittedAt()->getLocal() : ''
        );
    }
    elseif ($currentAssessable instanceof Task) {

        /** @var TaskCompletion $latestSubmission */
        $latestSubmission = null;
        $submittedBy = null;

        if ($currentAssessable->getAllResponses()->count()) {
            $latestSubmission = $currentAssessable->getAllResponses()->first();
            $submittedBy = $latestSubmission->getSubmittedBy();
        }

        new \Alaress_Probe(
            $currentAssessable->getTitle(),
            '',
            '',
            $submittedBy ? $submittedBy->getFullname() : '',
            '',
            $latestSubmission && $latestSubmission->getSubmittedAt() ? $latestSubmission->getSubmittedAt()->getLocal() : ''
        );
    }
}


echo "<h1>Feedbacks</h1>";

new \Alaress_Probe('Title', 'Files', 'Mark', 'Comment', 'Published At');

foreach ($project->getAssessables() as $currentAssessable) {
    if ($currentAssessable instanceof DueWork) {
        $allFiles = [];
        $allPlagscan = [];

        /** @var DueWorkSubmission $latestSubmission */
        $latestSubmission = null;

        /** @var DueWorkFeedback $latestFeedback */
        $latestFeedback = null;

        /** @var \Schoolbox\User $submittedBy */
        $submittedBy = null;

        /** @var \Schoolbox\User $submittedOnBehalfOf */
        $submittedOnBehalfOf = null;

        if ($currentAssessable->getAllResponses()->count()) {
            $latestSubmission = $currentAssessable->getAllResponses()->first();
            $submittedBy = $latestSubmission->getSubmittedBy();
            $submittedOnBehalfOf = $latestSubmission->getSubmittedOnBehalfOf();

            /** @var DueWorkFeedback $latestFeedback */
            $latestFeedback = $latestSubmission->findMostRecentNonNullFeedback();

            if ($latestFeedback) {
                /** @var DueWorkSubmissionFile $currentFile */
                foreach ($latestFeedback->getAllFiles() as $currentFile) {
                    $allFiles[] = $currentFile->getFilename();
                }
            }
        }

        new \Alaress_Probe(
            $currentAssessable->getTitle(),
            implode('<br />', $allFiles),
            $latestFeedback ? $latestFeedback->calculateMark() : '',
            $latestFeedback ? $latestFeedback->getComment() : '',
            $latestFeedback && $latestFeedback ? $latestFeedback->getPublishedAt()->getLocal() : 'Unpublished'
        );
    }
    elseif ($currentAssessable instanceof Quiz) {

        /** @var QuizAttempt $latestSubmission */
        $latestSubmission = null;

        /** @var QuizReview $latestFeedback */
        $latestFeedback = null;

        /** @var \Schoolbox\User $submittedBy */
        $submittedBy = null;

        if ($currentAssessable->getAllResponses()->count()) {
            $latestSubmission = $currentAssessable->getAllResponses()->first();
            $submittedBy      = $latestSubmission->getSubmittedBy();

            $latestFeedback = $latestSubmission->findMostRecentNonNullFeedback();
        }

        new \Alaress_Probe(
            $currentAssessable->getTitle(),
            implode('<br />', $allFiles),
            $latestFeedback ? $latestFeedback->calculateMark() : '',
            '',
            $latestFeedback && $latestFeedback->getPublishedAt() ? $latestFeedback->getPublishedAt()->getLocal() : 'Unpublished'
        );
    }
    elseif ($currentAssessable instanceof Task) {

        /** @var TaskCompletion $latestSubmission */
        $latestSubmission = null;

        /** @var TaskFeedback $latestFeedback */
        $latestFeedback = null;

        /** @var \Schoolbox\User $submittedBy */
        $submittedBy = null;

        $completionString = '';

        if ($currentAssessable->getAllResponses()->count()) {
            $latestSubmission = $currentAssessable->getAllResponses()->first();
            $submittedBy      = $latestSubmission->getSubmittedBy();

            /** @var TaskFeedback $latestFeedback */
            $latestFeedback = $latestSubmission->findMostRecentFeedback();

            if ($latestFeedback) {
                $completionString = 'Completion by ' . $latestSubmission->getSubmittedBy()->getFullname() . '<br />Feedback by ' . $latestFeedback->getReviewer()->getFullname();
            }
        }

        new \Alaress_Probe(
            $currentAssessable->getTitle(),
            '',
            $latestFeedback ?  $completionString : 'Not completed',
            '',
            $latestFeedback && $latestFeedback->getPublishedAt() ? $latestFeedback->getPublishedAt()->getLocal() : 'Unpublished'
        );
    }
}
