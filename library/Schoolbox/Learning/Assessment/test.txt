            $completionString = '';

            if ($currentParticipant->findAllResponsesForAssessable($currentAssessable)->count()) {
                $latestSubmission = $currentParticipant->findAllResponsesForAssessable($currentAssessable)->first();

                if ($latestSubmission) {
                    $submittedBy = $latestSubmission->getSubmittedBy();

                    $completionString = 'Completion by ' . $latestSubmission->getSubmittedBy()->getFullname();

                    /** @var TaskFeedback $latestFeedback */
                    $latestFeedback = $latestSubmission->findMostRecentFeedback();

                    if ($latestFeedback) {
                        $completionString .=  '<br />Feedback by ' . $latestFeedback->getReviewer()->getFullname();
                    }
                }
            }

            new \Alaress_Probe('<hr>');
            new \Alaress_Probe(
                "{$currentAssessable->getId()} {$currentAssessable->getTitle()}",
                $currentParticipant->getUser()->getId(),
                '',
                $completionString,
                '',
                $latestFeedback && $latestFeedback->isPublished() ? $latestFeedback->getPublishedAt()->getLocal() : 'Unpublished'
            );
        }
    }
}
