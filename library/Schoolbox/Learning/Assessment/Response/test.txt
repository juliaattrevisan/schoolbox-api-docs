<?php

header('Content-Type: text/html; charset=utf-8');

/** @var \Schoolbox\Application\WebRequest\Kernel $kernel */
$kernel = require_once(dirname(__FILE__) . '/../library/Bootstrap.php');

/** @var \Schoolbox\Learning\Assessment\Response\DataSource $responseDataSource */
$responseDataSource = $kernel->getDependencyInjector()->get('responseDataSource');

/** @var \Schoolbox\Learning\Assessment\Participant\DataSource $participantDataSource */

/** @var \Schoolbox\Learning\Assessment\Feedback\DataSource $feedbackDataSource */

/** @var \Schoolbox\Learning\Assessment\DataModel\MapperProxy $assessmentDataSource */
$assessmentDataSource = $kernel->getDependencyInjector()->get('assessmentDataSource');


echo '<h1>Due Work #268</h1>';
echo '<h2>Participants</h2>';
$dueWork = $assessmentDataSource->findById(268);
$dueWork->setAllParticipants($participantDataSource->findForAssessment($dueWork));
foreach ($dueWork->getAllParticipants() as $participant) {
    var_dump($participant->getUser()->formatName());

    echo '<h3>Responses</h3>';
    foreach ($participant->getAllResponses() as $response) {
        /** @var Schoolbox\Learning\Assessment\DueWork\Submission\Model $response */
        var_dump([
            get_class($response),
            $response->getId(),
            $response->getSubmittedBy()->formatName(),
            $response->getCreatedAt()->getDatabase(),
            $response->getAllFiles()->map(function($b) {return $b->getHash();}),
        ]);

        echo '<h4>Targeted By Feedback</h4>';
        var_dump($response->getAllFeedback()->map(function($v) {return $v->getId();})->getAll());
    }

    echo '<h3>Feedback</h3>';
    foreach ($participant->getAllFeedbacks() as $feedback) {
        /** @var Schoolbox\Learning\Assessment\DueWork\Feedback\Model $feedback */
        var_dump([
            get_class($feedback),
            $feedback->getId(),
            $feedback->getReviewee()->formatName(),
            $feedback->getReviewer()->formatName(),
            $feedback->getCreatedAt()->getDatabase(),
            $feedback->getNormalisedMark(),
            $feedback->getHumanReadableMark(),
            $feedback->getComment(),
        ]);

        echo '<h4>Targeted To Response</h4>';
        var_dump(
            ($feedback->getResponse() instanceof \Schoolbox\Learning\Assessment\IResponse
                ? $feedback->getResponse()->getId()
                : null
            )
        );
    }
}
