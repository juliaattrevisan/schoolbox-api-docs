<?php

header('Content-Type: text/html; charset=utf-8');

/** @var \Schoolbox\Application\WebRequest\Kernel $kernel */
$kernel = require_once(dirname(__FILE__) . '/../library/Bootstrap.php');

/** @var \Schoolbox\Learning\Assessment\Participant\DataSource $participantDataSource */
$participantDataSource = $kernel->getDependencyInjector()->get('participantDataSource');

/** @var \Schoolbox\Learning\Assessment\DataModel\MapperProxy $assessmentDataSource */
$assessmentDataSource = $kernel->getDependencyInjector()->get('assessmentDataSource');


/**
 * Dump the content of an assessment.
 *
 * @param \Schoolbox\Learning\Assessment\IAssessable $assessment
 */
function dumpAssessment(\Schoolbox\Learning\Assessment\IAssessable $assessment)
{
    // Here be HTML output
    ?>
    <h1>
        Assessment #<?= $assessment->getId() ?>
        (<a href="/homepage/<?= $assessment->getContext()->getFolder()->getId() ?>"><?= $assessment->getTypeName() ?>: <?= $assessment->getTitle() ?></a>)
    </h1>

    <table>
        <thead>
        <tr>
            <th>Participant</th>
            <th>Responses</th>
            <th>Feedback</th>
        </tr>
        </thead>
        <tbody>
        <?php foreach ($assessment->getAllParticipants() as $participant): ?>
            <tr>
                <td><?= $participant->getUser()->formatName() ?></td>
                <td>
                    <?php
                    foreach ($participant->getAllResponses() as $response) {
                        $responseDump = [
                            'class'   => get_class($response),
                            'generic' => [
                                'id'          => $response->getId(),
                                'submittedBy' => $response->getSubmittedBy()->formatName(),
                                'createdAt'   => $response->getCreatedAt()->getDatabase(),
                                'feedbackIds' => $response->getAllFeedback()
                                                          ->map(function ($v) { return $v->getId(); })
                                                          ->getAll(),
                            ]
                        ];
                        if ($response instanceof Schoolbox\Learning\Assessment\DueWork\Submission\Model) {
                            $responseDump['specific'] = [
                                'files' => $response->getAllFiles()
                                                    ->map(function($b) {return $b->getHash();})
                                                    ->getAll(),
                            ];
                        }
                        if ($response instanceof Schoolbox\Learning\Assessment\Quiz\Attempt\Model) {
                            $responseDump['specific'] = [
                                'marksGiven'     => $response->getMarksGiven(),
                                'marksAvailable' => $response->getMarksAvailable(),
                                'firstSectionId' => $response->getFirstSectionId(),
                            ];
                        }
                        var_dump($responseDump);
                    }
                    ?>
                </td>
                <td>
                    <?php
                    foreach ($participant->getAllFeedbacks() as $feedback) {
                        /** @var Schoolbox\Learning\Assessment\Quiz\Feedback\Model $feedback */
                        $feedbackDump = [
                            'class'   => get_class($feedback),
                            'generic' => [
                                'id'           => $feedback->getId(),
                                'reviewee'     => $feedback->getReviewee()->formatName(),
                                'reviewer'     => $feedback->getReviewer()->formatName(),
                                'createdAt'    => $feedback->getCreatedAt()->getDatabase(),
                                'normMark'     => $feedback->getNormalisedMark(),
                                'readableMark' => $feedback->getHumanReadableMark(),
                                'responseId'   => ($feedback->getResponse() instanceof \Schoolbox\Learning\Assessment\IResponse
                                    ? $feedback->getResponse()->getId()
                                    : null
                                ),
                            ]
                        ];
                        if ($feedback instanceof Schoolbox\Learning\Assessment\DueWork\Feedback\Model) {
                            $feedbackDump['specific'] = [
                                'comment' => $feedback->getComment(),
                            ];
                        }
                        var_dump($feedbackDump);
                    }
                    ?>
                </td>
            </tr>
        <?php endforeach; ?>
        </tbody>
    </table>
    <?php
    // Here end HTML output
}
?>
<style type="text/css">
    table {
        width: 100%;
        border: 1px solid black;
        border-collapse: collapse;
    }
    table tr {
        vertical-align: top;
    }
    table th {
        background-color: #e8f3e8;
        border-bottom: 1px solid black;
        padding: 0.5rem;
    }
    table td {
        border-bottom: 1px solid black;
        padding: 0.5rem;
    }
</style>
<?php

$dueWork = $assessmentDataSource->findById(268);
$dueWork->setAllParticipants($participantDataSource->findForAssessment($dueWork));
dumpAssessment($dueWork);


$task = $assessmentDataSource->findById(206);
$task->setAllParticipants($participantDataSource->findForAssessment($task));
dumpAssessment($task);


$quiz = $assessmentDataSource->findById(277);
$quiz->setAllParticipants($participantDataSource->findForAssessment($quiz));
dumpAssessment($quiz);
