/**
 * Shortcut for assert($value1 == $value2).
 * @param mixed $value1
 * @param mixed $value2
 */
function assertEquals($value1, $value2)
{
    $backtrace = debug_backtrace();
    $origin    = $backtrace[0];
    var_dump(sprintf('%d: %s(%s, %s)', $origin['line'], $origin['function'], $value1, $value2));
    assert($value1 == $value2);
}

// Mapper for assessables, and its required translator

    new \Schoolbox\Learning\MarkScheme\Factory(),
    new \Schoolbox\Learning\Course\DataModel\Specification\Unit\Translator(),
    new \Schoolbox\Learning\Course\DataModel\Implementation\Unit\Translator()
);

    ->addSubTranslator(new \Schoolbox\Learning\Assessment\DataModel\Translator\Task())

// Create a task
$task = (new Task())
    // Generic fields
    ->setContext($folder)
    ->setCreator((new \Schoolbox\User())->setId(3))
    ->setPublishAt(new Alaress_Date_Format('2016-10-14 12:00:00', Alaress_Date_Format::DATABASE_FORMAT))

    ->setDueAt(new Alaress_Date_Format('2016-10-18 12:00:00', Alaress_Date_Format::DATABASE_FORMAT))
    ->setCloseAt(new Alaress_Date_Format('2016-10-23 13:00:00', Alaress_Date_Format::DATABASE_FORMAT))
    ->setTitle('Mr Test Task сказать')
    // Description is ignored for tasks
    ->setDescription('Blaargh ничего')
    // Tasks are always weight 0, so this doesn't matter
    ->setWeight(1000)
    // Specific fields
    ->setUrl('https://www.something.com/')
    ->selfAssessed()
;

$mapper->save($task, 0);

// Change and save it again
$task->teacherAssessed()
     ->setUrl('https://www.nothing.com/')
;
$mapper->save($task, 0);

// Load from DB and assert that all fields match
/** @var Task $dbTask */
$dbTask = $mapper->findById($task->getId());
// Generic fields
assertEquals($dbTask->getId(), $task->getId());
assertEquals($dbTask->getCreator()->getId(), $task->getCreator()->getId());
assertEquals($dbTask->getPublishAt()->getUnixTimeStamp(), $task->getPublishAt()->getUnixTimeStamp());
assertEquals($dbTask->getOpenAt()->getUnixTimeStamp(), $task->getOpenAt()->getUnixTimeStamp());
assertEquals($dbTask->getDueAt()->getUnixTimeStamp(), $task->getDueAt()->getUnixTimeStamp());
assertEquals($dbTask->getCloseAt()->getUnixTimeStamp(), $task->getCloseAt()->getUnixTimeStamp());
assertEquals($dbTask->getTitle(), $task->getTitle());
assertEquals($dbTask->getDescription(), $task->getDescription());
assertEquals($dbTask->getWeight(), 0);
// Context
assertEquals(get_class($dbTask->getContext()), get_class($task->getContext()));
assertEquals($dbTask->getContext()->getId(), $task->getContext()->getId());
// Specific fields
assertEquals($dbTask->getUrl(), $task->getUrl());
assertEquals($dbTask->isSelfAssessed(), $task->isSelfAssessed());
assertEquals($dbTask->isTeacherAssessed(), $task->isTeacherAssessed());

// Delete and try to refetch it
$mapper->delete($task);
try {
    $dbTask = $mapper->findById($task->getId());
    // This must have thrown an exception
    assert(false, 'Task should have been deleted');
} catch (\Schoolbox\Mapper\Exception\Deleted $e) {
    assert($e->getDeletedObject() instanceof Task);
    assertEquals($dbTask->getId(), $task->getId());
}
