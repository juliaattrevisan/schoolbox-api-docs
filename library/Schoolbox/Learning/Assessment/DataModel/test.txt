/**
 * Shortcut for assert($value1 == $value2).
 * @param mixed $value1
 * @param mixed $value2
 */
function assertEquals($value1, $value2) {
    assert($value1 == $value2, sprintf('"%s" != "%s"', $value1, $value2));
}

// Mapper for assessables, and its required translator
$translator = new \Schoolbox\Learning\Assessment\DataModel\Translator(
    new \Schoolbox\Learning\MarkScheme\Factory()
);
$translator
    ->addSubTranslator(new \Schoolbox\Learning\Assessment\DataModel\Translator\Task())
    ->addSubTranslator(new \Schoolbox\Learning\Assessment\DataModel\Translator\DueWork())
    ->addSubTranslator(new \Schoolbox\Learning\Assessment\DataModel\Translator\Quiz())
    ->addSubTranslator(new \Schoolbox\Learning\Assessment\DataModel\Translator\Project($translator))
;
$mapper = new \Schoolbox\Learning\Assessment\DataModel\MapperProxy(
    \Schoolbox\Learning\Assessment\DataModel\Mapper::getInstance(),
    $translator
);

// Folder
$folder = Schoolbox\Resource\Folder\Mapper::getInstance()->findById(51);
// Unit spec and impl: there is at least one of these in demo data.
// That is good enough for this test.
$unitSpec = (new \Schoolbox\Learning\Course\Unit\Specification())->setId(1);
$unitImpl = (new \Schoolbox\Learning\Course\Unit\Implementation())->setId(1);


/**
 * Tasks
 */

// Create a task
$task = (new Task())
    // Generic fields
    ->setAssessmentContext($folder)
    ->setCreator((new \Schoolbox\User())->setId(3))
    ->setPublishAt(new Alaress_Date_Format('2016-10-14 12:00:00', Alaress_Date_Format::DATABASE_FORMAT))
    ->setOpenAt(new Alaress_Date_Format('2016-10-16 12:00:00', Alaress_Date_Format::DATABASE_FORMAT))
    ->setDueAt(new Alaress_Date_Format('2016-10-18 12:00:00', Alaress_Date_Format::DATABASE_FORMAT))
    ->setCloseAt(new Alaress_Date_Format('2016-10-23 13:00:00', Alaress_Date_Format::DATABASE_FORMAT))
    ->setTitle('Mr Test Task сказать')
    // Description is ignored for tasks
    ->setDescription('Blaargh ничего')
    // Tasks are always weight 0, so this doesn't matter
    ->setWeight(1000)
    // Specific fields
    ->setLink('https://www.something.com/')
    ->setSelfMarkable(true)
;

$mapper->save($task, 0);

// Change and save it again
$task->setSelfMarkable(false)
     ->setLink('https://www.nothing.com/')
;
$mapper->save($task, 0);

// Load from DB and assert that all fields match
/** @var Task $dbTask */
$dbTask = $mapper->findById($task->getId());
// Generic fields
assertEquals($dbTask->getId(), $task->getId());
assertEquals($dbTask->getCreator()->getId(), $task->getCreator()->getId());
assertEquals($dbTask->getPublishAt()->getUnixTimeStamp(), $task->getPublishAt()->getUnixTimeStamp());
assertEquals($dbTask->getOpenAt()->getUnixTimeStamp(), $task->getOpenAt()->getUnixTimeStamp());
assertEquals($dbTask->getDueAt()->getUnixTimeStamp(), $task->getDueAt()->getUnixTimeStamp());
assertEquals($dbTask->getCloseAt()->getUnixTimeStamp(), $task->getCloseAt()->getUnixTimeStamp());
assertEquals($dbTask->getTitle(), $task->getTitle());
assertEquals($dbTask->getDescription(), $task->getDescription());
assertEquals($dbTask->getWeight(), 0);
// Context
assertEquals(get_class($dbTask->getAssessmentContext()), get_class($task->getAssessmentContext()));
assertEquals($dbTask->getAssessmentContext()->getId(), $task->getAssessmentContext()->getId());
// Specific fields
assertEquals($dbTask->getLink(), $task->getLink());
assertEquals($dbTask->isSelfMarkable(), $task->isSelfMarkable());

// Delete and try to refetch it
$mapper->delete($task);
try {
    $dbTask = $mapper->findById($task->getId());
    // This must have thrown an exception
    assert(false, 'Task should have been deleted');
} catch (\Schoolbox\Mapper\Exception\Deleted $e) {
    assert($e->getDeletedObject() instanceof Task);
    assertEquals($dbTask->getId(), $task->getId());
}

/**
 * Due Works

// Preliminary step: upload some files into storage
$storageMapper = \Schoolbox\Resource\Storage\Mapper::getInstance();
$file1         = $storageMapper->acceptStream('test1.txt', 'a test file');
$file2         = $storageMapper->acceptStream('test2.txt', 'not a test');
$file3         = $storageMapper->acceptStream('test3.txt', 'also not a test');

// Create a due work
$dueWork = (new DueWork())
    // Generic fields
    ->setAssessmentContext($unitImpl)
    ->setCreator((new \Schoolbox\User())->setId(3))
    ->setPublishAt(new Alaress_Date_Format('2016-10-14 12:00:00', Alaress_Date_Format::DATABASE_FORMAT))
    ->setOpenAt(new Alaress_Date_Format('2016-10-16 12:00:00', Alaress_Date_Format::DATABASE_FORMAT))
    ->setDueAt(new Alaress_Date_Format('2016-10-18 12:00:00', Alaress_Date_Format::DATABASE_FORMAT))
    ->setCloseAt(new Alaress_Date_Format('2016-10-23 13:00:00', Alaress_Date_Format::DATABASE_FORMAT))
    ->setTitle('Due Work Test งาน')
    ->setDescription('Something ไม่มีอะไร')
    ->setWeight(21)
    // Specific fields
    ->setAcceptsMultiple()
    ->showOnlySubmittedStudents()
    ->setAllFiles((new \Schoolbox\Collection())
        // Duework files have no separate filenames
        ->add((new DueWorkFile())
            ->setFilename('test1.txt')
            ->setHash($file1->getHash())
        )
        ->add((new DueWorkFile())
            ->setFilename('test2.txt')
            ->setHash($file2->getHash())
        )
        ->add((new DueWorkFile())
            ->setFilename('test3.txt')
            ->setHash($file3->getHash())
        )
    )
    ->setType((new DueWorkType())
        ->setId(12)
        ->setName('A dumb type')
    )
    ->setMarkScheme(new \Schoolbox\Learning\MarkScheme\TotalMarkWholeNumber(40))
    ->setHasRubric(true)
;
// Due work files have no sequence stored, so order by name
$dueWork->getAllFiles()->sortBy('filename', 'asc');
$mapper->save($dueWork, 0);

// Change and save it again
$dueWork->setWeight(3);
$mapper->save($dueWork, 0);

// Load from DB and assert that all fields match
/** @var DueWork $dbDueWork */
$dbDueWork = $mapper->findById($dueWork->getId());
// Generic fields
assertEquals($dbDueWork->getId(), $dueWork->getId());
assertEquals($dbDueWork->getCreator()->getId(), $dueWork->getCreator()->getId());
assertEquals($dbDueWork->getPublishAt()->getUnixTimeStamp(), $dueWork->getPublishAt()->getUnixTimeStamp());
assertEquals($dbDueWork->getOpenAt()->getUnixTimeStamp(), $dueWork->getOpenAt()->getUnixTimeStamp());
assertEquals($dbDueWork->getDueAt()->getUnixTimeStamp(), $dueWork->getDueAt()->getUnixTimeStamp());
assertEquals($dbDueWork->getCloseAt()->getUnixTimeStamp(), $dueWork->getCloseAt()->getUnixTimeStamp());
assertEquals($dbDueWork->getTitle(), $dueWork->getTitle());
assertEquals($dbDueWork->getDescription(), $dueWork->getDescription());
assertEquals($dbDueWork->getWeight(), $dueWork->getWeight());
// Context
assertEquals(get_class($dbDueWork->getAssessmentContext()), get_class($dueWork->getAssessmentContext()));
assertEquals($dbDueWork->getAssessmentContext()->getId(), $dueWork->getAssessmentContext()->getId());
// Specific fields
assertEquals($dbDueWork->isAcceptSingle(), $dueWork->isAcceptSingle());
assertEquals($dbDueWork->isAcceptNone(), $dueWork->isAcceptNone());
assertEquals($dbDueWork->isAcceptMulti(), $dueWork->isAcceptMulti());
assertEquals($dbDueWork->getType()->getId(), $dueWork->getType()->getId());
assertEquals(get_class($dbDueWork->getMarkScheme()), get_class($dueWork->getMarkScheme()));
assertEquals($dbDueWork->getMarkScheme()->getTotalMark(), $dueWork->getMarkScheme()->getTotalMark());
// Due work files
assertEquals($dbDueWork->getAllFiles()->count(), $dueWork->getAllFiles()->count());
// Sequence isn't stored for due work files
$dbDueWork->getAllFiles()->sortBy('filename', 'asc');
assertEquals($dbDueWork->getAllFiles()[0]->getFilename(), $dueWork->getAllFiles()[0]->getFilename());
assertEquals($dbDueWork->getAllFiles()[1]->getFilename(), $dueWork->getAllFiles()[1]->getFilename());
assertEquals($dbDueWork->getAllFiles()[2]->getFilename(), $dueWork->getAllFiles()[2]->getFilename());

// Delete and try to refetch it
$mapper->delete($dueWork);
try {
    $dbDueWork = $mapper->findById($dueWork->getId());
    // This must have thrown an exception
    assert(false, 'Due work should have been deleted');
} catch (\Schoolbox\Mapper\Exception\Deleted $e) {
    assert($e->getDeletedObject() instanceof DueWork);
    assertEquals($dbDueWork->getId(), $dueWork->getId());
}
