<?php

// Copy this file to /www/test.php, then access /test.php on your vagrant box.

require_once __DIR__ . '/../library/Bootstrap.php';


// Create a Course Specification for a Folder
$courseSpecFolder = new Schoolbox\Resource\Folder\Model();
$courseSpecFolder->setId(51)->setName('Houses');
$courseSpec       = new Schoolbox\Learning\Course\Specification\Model();
$courseSpec       = $courseSpec->setFolder($courseSpecFolder);
// Add the Course Spec back to the Folder
$courseSpecFolder = $courseSpecFolder->setCourseSpecification($courseSpec);


// Create Unit Specifications for all the children of that Folder

// Unit Spec: Fox House
$foxSpecFolder = new Schoolbox\Resource\Folder\Model();
$foxSpecFolder
    ->setId(52)
    ->setName('Fox House');
$foxSpec = new Schoolbox\Learning\Course\Specification\Unit\Model();
$foxSpec->setFolder($foxSpecFolder);

// Fox House Assessables
$foxDueWork1 = new Schoolbox\Learning\Assessment\DueWork\Model();
$foxDueWork1
    ->setId(1)
    ->setType((new \Schoolbox\Learning\Assessment\DueWork\Type())->setId(1)->setName('Essay'))
    ->setTitle('Fox House Due Work')
    ->setAcceptsMultiple()
    ->showOnlySubmittedStudents();

$foxDueWork2 = new Schoolbox\Learning\Assessment\DueWork\Model();
$foxDueWork2
    ->setId(2)
    ->setType((new \Schoolbox\Learning\Assessment\DueWork\Type())->setId(2)->setName('Homework'))
    ->setTitle('Fox House Due Work 2')
    ->setAcceptsMultiple()
    ->showOnlySubmittedStudents();

$foxSpec->getAssessables()
        ->add($foxDueWork1)
        ->add($foxDueWork2);


// Unit Spec: Oak House
$oakSpecFolder = new Schoolbox\Resource\Folder\Model();
$oakSpecFolder
    ->setId(54)
    ->setName('Oak House');
$oakSpec = new Schoolbox\Learning\Course\Specification\Unit\Model();
$oakSpec->setFolder($oakSpecFolder);

// Oak House Assessables
$oakTask1 = new Schoolbox\Learning\Assessment\Task\Model();
$oakTask1
    ->setId(1)
    ->setTitle('Oak Task 1');
$oakTask2 = new Schoolbox\Learning\Assessment\Task\Model();
$oakTask2
    ->setId(2)
    ->setTitle('Oak Task 2');

$oakSpec->getAssessables()
        ->add($oakTask1)
        ->add($oakTask2);


// Unit Spec: Thistle House
$thistleSpecFolder = new Schoolbox\Resource\Folder\Model();
$thistleSpecFolder
    ->setId(55)
    ->setName('Thistle House');
$thistleSpec = new Schoolbox\Learning\Course\Specification\Unit\Model();
$thistleSpec->setFolder($thistleSpecFolder);

// Thistle House Assessables
$thistleQuiz1 = new Schoolbox\Learning\Assessment\Quiz\Model();
$thistleQuiz1
    ->setId(1)
    ->setTitle('Thistle Quiz 1');
$thistleQuiz2 = new Schoolbox\Learning\Assessment\Quiz\Model();
$thistleQuiz2
    ->setId(2)
    ->setTitle('Thistle Quiz 2');

$thistleSpec->getAssessables()
            ->add($thistleQuiz1)
            ->add($thistleQuiz2);


// Unit Spec: Dalton House
$daltonSpecFolder = new Schoolbox\Resource\Folder\Model();
$daltonSpecFolder
    ->setId(56)
    ->setName('Dalton House');
$daltonSpec = new Schoolbox\Learning\Course\Specification\Unit\Model();
$daltonSpec->setFolder($daltonSpecFolder);

// Dalton House Assessables
$daltonTask = new Schoolbox\Learning\Assessment\Task\Model();
$daltonTask
    ->setId(3)
    ->setTitle('Dalton Task');

$daltonQuiz = new Schoolbox\Learning\Assessment\Quiz\Model();
$daltonQuiz
    ->setId(3)
    ->setTitle('Dalton Quiz');

$daltonDueWork = new Schoolbox\Learning\Assessment\DueWork\Model();
$daltonDueWork
    ->setId(3)
    ->setType((new \Schoolbox\Learning\Assessment\DueWork\Type())->setId(3)->setName('Assessment Task'))
    ->setTitle('Dalton Due Work')
    ->setAcceptsMultiple()
    ->showOnlySubmittedStudents();

$daltonSpec->getAssessables()
           ->add($daltonTask)
           ->add($daltonDueWork)
           ->add($daltonQuiz);


// Add all unit specs to course spec
$courseSpec->getUnits()
           ->add($foxSpec)
           ->add($oakSpec)
           ->add($thistleSpec)
           ->add($daltonSpec);


// Create Course Imports for a few class Folders
$frenchClass = new Schoolbox\Resource\Folder\Model();
$frenchClass->setId(237)->setName('9AH French');
$frenchClassCourseImport = new Schoolbox\Learning\Course\Import\Model();
$frenchClassCourseImport = $frenchClassCourseImport
    ->setFolder($frenchClass)
    ->setSpecification($courseSpec);

$englishClass = new Schoolbox\Resource\Folder\Model();
$englishClass->setId(43)->setName('English 3 4');
$englishClassCourseImport = new Schoolbox\Learning\Course\Import\Model();
$englishClassCourseImport = $englishClassCourseImport
    ->setFolder($englishClass)
    ->setSpecification($courseSpec);

$scienceClass = new Schoolbox\Resource\Folder\Model();
$scienceClass->setId(357)->setName('Science Unit 2');
$scienceClassCourseImport = new Schoolbox\Learning\Course\Import\Model();
$scienceClassCourseImport = $scienceClassCourseImport
    ->setFolder($scienceClass)
    ->setSpecification($courseSpec);


// Add the Course Imports back to the Spec Folder
$courseSpecFolder = $courseSpecFolder->setCourseImports(
    $courseSpecFolder->getCourseImports()
                     ->add($frenchClassCourseImport)
                     ->add($englishClassCourseImport)
                     ->add($scienceClassCourseImport)
);


// Create Unit Imports for the Unit Specs in the Course Spec, and add them to each class.
// French: only add even numbered unit specs
// English: only add odd numbered unit specs
// Science: add all unit specs
$i = 1;
foreach ($courseSpec->getUnits() as $unitSpec) {
    $title = $i . ': ' . $unitSpec->getFolder()->getName();
    if ($i % 2 == 0) {
        $frenchClassUnitImport = new Schoolbox\Learning\Course\Import\Unit\Model();
        $frenchClassUnitImport
            ->setTitle($title)
            ->setSpecification($unitSpec)
            ->setOpenDate(Alaress_Date_Format::now()->startOfDay()->addMonths($i)->startOfMonth())
            ->setDueDate(Alaress_Date_Format::now()->startOfDay()->addMonths($i)->endOfMonth())
            ->populateAssessablesFromSpecification($unitSpec);
        $frenchClassCourseImport->getUnits()->add($frenchClassUnitImport);
    } else {
        $englishClassUnitImport = new Schoolbox\Learning\Course\Import\Unit\Model();
        $englishClassUnitImport
            ->setTitle($title)
            ->setSpecification($unitSpec)
            ->setOpenDate(Alaress_Date_Format::now()->startOfDay()->addMonths($i)->startOfMonth()->addDays(3))
            ->setDueDate(Alaress_Date_Format::now()->startOfDay()->addMonths($i)->endOfMonth()->addDays(3))
            ->populateAssessablesFromSpecification($unitSpec);
        $englishClassCourseImport->getUnits()->add($englishClassUnitImport);
    }
    $scienceClassUnitImport = new Schoolbox\Learning\Course\Import\Unit\Model();
    $scienceClassUnitImport
        ->setTitle($title)
        ->setSpecification($unitSpec)
        ->setOpenDate(Alaress_Date_Format::now()->startOfDay()->addMonths($i)->startOfMonth()->addDays(13))
        ->setDueDate(Alaress_Date_Format::now()->startOfDay()->addMonths($i)->endOfMonth()->addDays(13))
        ->populateAssessablesFromSpecification($unitSpec);
    $scienceClassCourseImport->getUnits()->add($scienceClassUnitImport);
    $i++;
}


// Augmented/Diminished/Reordered Course Imports for Science Class
foreach ($scienceClassCourseImport->getUnits() as $unit) {
    switch ($unit->getSpecification()->getFolder()->getId()) {
        case 52:
            // Fox House for Science Class:
            // * Remove first assessable
            // * Add another
            $augmentTask = new Schoolbox\Learning\Assessment\Task\Model();
            $augmentTask->setTitle('Science-only Task');
            $unit->getAssessables()->slice(1)->add($augmentTask);
            break;

        case 55:
            // Thistle House for Science Class:
            // * Reorder assessables
            $unit->getAssessables()->sortBy('title', 'desc');
            break;

        default:
            break;
    }
}
